
package com.company.automation.smoke;

import io.appium.java_client.windows.WindowsDriver;
import org.openqa.selenium.remote.DesiredCapabilities;
import org.testng.Assert;
import org.testng.annotations.Test;

import java.net.URL;

public class DesktopSmokeTest {

    @Test
    public void shouldLaunchWindowsNotepad() throws Exception {
        WindowsDriver driver = null;

        try {
            DesiredCapabilities capabilities =
                    new DesiredCapabilities();

            capabilities.setCapability("platformName", "Windows");
            capabilities.setCapability("deviceName", "WindowsPC");
            capabilities.setCapability(
                    "app",
                    "C:\\Windows\\System32\\notepad.exe"
            );

            driver = new WindowsDriver(
                    new URL("http://127.0.0.1:4723"),
                    capabilities
            );

            String title = driver.getTitle();

            System.out.println("Notepad window title: " + title);

            Assert.assertTrue(
                    title.toLowerCase().contains("notepad"),
                    "Notepad did not launch as expected: " + title
            );

        } finally {
            if (driver != null) {
                driver.quit();
            }
        }
    }
}
